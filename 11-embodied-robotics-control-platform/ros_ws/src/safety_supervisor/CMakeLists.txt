cmake_minimum_required(VERSION 3.8)
project(safety_supervisor)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(embodied_interfaces REQUIRED)
include_directories(include)
add_executable(safety_supervisor src/safety_supervisor.cpp)
ament_target_dependencies(safety_supervisor rclcpp embodied_interfaces)
install(TARGETS safety_supervisor DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_safety_rules test/test_safety_rules.cpp)
endif()
ament_package()
